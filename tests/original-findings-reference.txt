import assert from 'node:assert/strict';import fs from 'node:fs';import {loadCore,node,prop,snapshot} from './core-probe.mjs';
const c=await loadCore();let assertions=0;const check=(v,message)=>{assert.ok(v,message);assertions++;};const root=node(1,-1,'RootWebArea','AX test');
const s=snapshot([root,node(2,0,'button','A'),node(3,0,'button','B')]);const a=c.build(s);
check(a.text.includes('\n\t1 button A\n'),'tab indentation');check(a.identity(1)==='1:1|0:|1:2','length-prefixed identity');check(a.identity(99)===null,'missing identity');
let t=structuredClone(s);t.nodes[1].name='Renamed';let b=c.build(t,a);check(b.identity(1)===a.identity(1),'rename preserves identity');
t=structuredClone(s);t.nodes.splice(1,0,node(10,0,'button','Inserted'));b=c.build(t,a);check(b.text.includes('3 button Inserted'),'new id uses previous maximum');check(b.text.includes('1 button A'),'insert preserves old ids');
t=structuredClone(s);[t.nodes[1],t.nodes[2]]=[t.nodes[2],t.nodes[1]];b=c.build(t,a);check(b.text.includes('2 button B'),'move B retains id');check(b.text.includes('3 button A'),'reorder can assign new id');
t=structuredClone(s);t.tab.id=2;assert.throws(()=>c.build(t,a),/status=5/);assertions++;
const big=snapshot([root,...Array.from({length:100},(_,i)=>node(i+2,0,'button',`Action ${i} ${'a'.repeat(60)}`))]);const p=c.build(big);b=c.build(big,p);check(b.text==='There has been no change in the accessibility tree.','nochange text');
t=structuredClone(big);t.nodes[2].name='Renamed';b=c.build(t,p);check(b.text.startsWith('The following is a diff'),'auto diff');check(b.text.includes('~\t2 button Renamed'),'changed marker');const full=c.build(t,p,true);check(full.text.startsWith('0 AXWebArea'),'forced full');check(full.identity(2)===p.identity(2),'full does not reset ids');
t.nodes.splice(4,2);t.nodes.push(node(900,0,'button','Added'));b=c.build(t,p);check(b.text.includes('Removed element IDs: 4-5'),'removed range');check(b.text.includes('+\t101 button Added'),'added marker');
for(const len of [986,987]){const v=snapshot([node(1,-1,'RootWebArea','Test'),node(2,0,'button','a'.repeat(len))]);const q=c.build(v),r=c.build(v,q);check(r.text.startsWith('There has')===(len===987),'1000 byte saved boundary');}

const many=snapshot([root,...Array.from({length:700},(_,i)=>node(i+2,0,'button','B'+i))]);b=c.build(many);check(b.text.includes('showing 0-500 of 700 items'),'sibling subset');
const off=snapshot([root,node(2,0,'textbox','Query',{value:'hello',properties:{focused:prop(true),editable:prop('plaintext')},dom:{tagName:'input',bounds:[0,10000,100,20]}})]);b=c.build(off);check(b.text.includes('hello'),'offscreen input included');check(b.text.includes('The focused UI element is'),'focused footer');
